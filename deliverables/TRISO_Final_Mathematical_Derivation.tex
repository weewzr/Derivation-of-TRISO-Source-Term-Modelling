\section{Scope and assumptions}
\begin{enumerate}
\item Five concentric regions: Kernel, Buffer, IPyC, SiC and OPyC. Region $i$ occupies $R_{i-1}<r<R_i$.
\item Spherical symmetry. In each region $D_i$ is constant.
\item Interfaces are ideal: concentration and normal diffusive flux are continuous (no partition coefficient).
\item \textbf{Case A (release).} Frozen initial kernel inventory, no source for $t>0$, absorbing outer surface. The finite-volume (FV) solver and the exact series of Section~\ref{sec:exact} both solve Case~A with identical inputs, so they are compared directly.
\item \textbf{Case B (analytical benchmark).} Continuous generation with a Robin outer boundary. Only its homogeneous eigenvalue condition is implemented (Section~\ref{sec:robin}). Case~A and Case~B must not be compared with each other.
\end{enumerate}

\section{From conservation to the spherical equation}
For any fixed control volume $V$ with outward unit normal $\mathbf n$,
\begin{equation}
\frac{d}{dt}\int_V c\,dV=-\int_{\partial V}\mathbf J\cdot\mathbf n\,dA+\int_V S\,dV.
\tag{TRISO-GOV-020}
\end{equation}
The divergence theorem turns the surface term into a volume integral,
\[
\int_V\left(\frac{\partial c}{\partial t}+\nabla\cdot\mathbf J-S\right)dV=0 ,
\]
and because $V$ is arbitrary the integrand vanishes:
\begin{equation}
\frac{\partial c}{\partial t}+\nabla\cdot\mathbf J=S.
\tag{TRISO-GOV-030}
\end{equation}
Fick's law
\begin{equation}
\mathbf J=-D\nabla c
\tag{TRISO-GOV-034}
\end{equation}
gives
\begin{equation}
\frac{\partial c}{\partial t}=\nabla\cdot(D\nabla c)+S.
\tag{TRISO-GOV-040}
\end{equation}
With spherical symmetry $\mathbf J=J_r(r,t)\,\mathbf e_r$. A balance on the shell $r<r'<r+dr$ gives
$\partial_t(4\pi r^2c\,dr)=-\left[4\pi(r+dr)^2J_r(r+dr)-4\pi r^2J_r(r)\right]+4\pi r^2S\,dr$, so $\nabla\cdot\mathbf J=r^{-2}\partial_r(r^2J_r)$ and, in region $i$,
\begin{equation}
\boxed{\frac{\partial c_i}{\partial t}=\frac{1}{r^2}\frac{\partial}{\partial r}\left(r^2D_i\frac{\partial c_i}{\partial r}\right)+S_i}
\tag{TRISO-FV-100}
\end{equation}

\section{Conditions}
Centre (regularity and symmetry):
\[
\left.\frac{\partial c_1}{\partial r}\right|_{r=0}=0 .
\]
Interfaces $r=R_i$, $i=1,\dots,4$ (two conditions each):
\[
c_i(R_i,t)=c_{i+1}(R_i,t),\qquad D_i\left.\frac{\partial c_i}{\partial r}\right|_{R_i^-}=D_{i+1}\left.\frac{\partial c_{i+1}}{\partial r}\right|_{R_i^+}.
\]
Outer surface, Case A (absorbing) and Case B (Robin):
\[
c_5(R,t)=0\qquad\text{or}\qquad -D_5\left.\frac{\partial c_5}{\partial r}\right|_R=h\,[c_5(R,t)-c_\infty].
\]
Initial data and source, Case A:
\[
c(r,0)=\begin{cases}c_0,&0\le r<R_1\\0,&R_1<r<R\end{cases}\qquad S_i=0\ (t>0).
\]

\section{Exact solution of Case A}\label{sec:exact}
\subsection{Separation and transformation}
Set $c=\phi(r)\,e^{-\Lambda t}$. Region by region (the product $r^2D\,\phi'$ is the conserved flux quantity),
\[
\frac{1}{r^2}\frac{d}{dr}\left(r^2D_i\frac{d\phi}{dr}\right)+\Lambda\phi=0 .
\]
With $u=r\phi$ (Equation TRISO-ML-431) this becomes, in each region,
\begin{equation}
u''+k_i^2u=0,\qquad k_i^2=\frac{\Lambda}{D_i},
\tag{TRISO-ML-432}
\end{equation}
\begin{equation}
u_i(r)=A_i\sin(k_ir)+B_i\cos(k_ir),\qquad \phi_i=\frac{u_i}{r}.
\tag{TRISO-ML-433}
\end{equation}
Regularity at the centre removes $B_1\cos(k_1r)/r$, so $B_1=0$ (TRISO-ML-435).

\subsection{Conditions in the transformed variable}
Concentration continuity $\phi_i(R_i)=\phi_{i+1}(R_i)$ gives, after multiplying by $R_i$,
\begin{equation}
u_i(R_i)=u_{i+1}(R_i).
\tag{TRISO-ML-439}
\end{equation}
Flux continuity uses $\phi'=(u'-u/r)/r$:
\begin{equation}
D_i\left[u_i'(R_i)-\frac{u_i(R_i)}{R_i}\right]=D_{i+1}\left[u_{i+1}'(R_i)-\frac{u_{i+1}(R_i)}{R_i}\right].
\tag{TRISO-ML-446}
\end{equation}
The outer condition for Case~A is $u_5(R)=0$.

\textbf{Caution.} The quantity $D\,u'$ is not continuous across an interface when the diffusivities differ. Matching $D\,u'$ instead of Equation TRISO-ML-446 gives wrong eigenvalues and wrong release curves.

\subsection{Characteristic function}
Carry the state $(u,q)$ with $q=D\,u'$ from the centre outward. Inside a region of thickness $L$ (solution of Equation TRISO-ML-432),
\begin{equation}
\begin{pmatrix}u\\ q\end{pmatrix}_{\mathrm{out}}=\begin{pmatrix}\cos kL & \dfrac{\sin kL}{Dk}\\[2mm] -Dk\sin kL & \cos kL\end{pmatrix}\begin{pmatrix}u\\ q\end{pmatrix}_{\mathrm{in}}.
\tag{TRISO-DEL-010}
\end{equation}
Across the interface at $R_i$, Equation TRISO-ML-439 keeps $u$ and Equation TRISO-ML-446 gives
\begin{equation}
q_{i+1}(R_i)=q_i(R_i)+\left(D_{i+1}-D_i\right)\frac{u(R_i)}{R_i}.
\tag{TRISO-DEL-011}
\end{equation}
Start from $(u,q)=(0,1)$ at $r=0$ (regular solution), propagate through all five regions, and define
\begin{equation}
g(\Lambda)=u(R;\Lambda),\qquad g(\Lambda_n)=0.
\tag{TRISO-DEL-012}
\end{equation}
The roots of $g$ are the eigenvalues. This is the same condition as $\det\mathbf M(\Lambda)=0$ (TRISO-ML-474) for the global coefficient system, evaluated by propagation instead of by a determinant.

\textbf{Completeness of the root list.} The number of eigenvalues not exceeding $\Lambda$ equals the number of zeros of the regular solution in $(0,R]$ (Sturm oscillation count). Isolating roots by bisection on this integer count cannot skip a root.

\subsection{Orthogonality and release series}
With weight $r^2$ for $\phi$, that is weight $1$ for $u$, the interface and boundary terms cancel and (TRISO-ML-528, TRISO-ML-531)
\[
\int_0^R u_mu_n\,dr=0\qquad(m\ne n).
\]
Expanding the initial data, $c=\sum_nA_n\phi_ne^{-\Lambda_nt}$ with $A_n=c_0\,I^{(1)}_n/N_n$, where
\begin{equation}
I^{(1)}_n=\int_0^{R_1}r\,u_n\,dr,\qquad I_n=\int_0^{R}r\,u_n\,dr,\qquad N_n=\int_0^{R}u_n^2\,dr.
\tag{TRISO-DEL-020}
\end{equation}
The inventory is $M(t)=4\pi\sum_nA_nI_ne^{-\Lambda_nt}$ and $M^0=\tfrac{4\pi}{3}c_0R_1^3$. Hence
\begin{equation}
\begin{gathered}
F(t)=1-\sum_{n}w_n\,e^{-\Lambda_nt},\\
w_n=\frac{3\,I^{(1)}_nI_n}{R_1^3\,N_n},\qquad\sum_nw_n=1.
\end{gathered}
\tag{TRISO-DEL-021}
\end{equation}
The integrals are evaluated region by region from the propagated states, using $u(r)=u\cos ks+(q/Dk)\sin ks$ with $s=r-R_{i-1}$.

\textbf{Homogeneous check.} For equal $D$, $u_n=\sin(n\pi r/R)$, $\Lambda_n=n^2\pi^2D/R^2$ and $N_n=R/2$. When the initial data fill the whole sphere ($R_1=R$) this gives $w_n=6/(n^2\pi^2)$, the classical result $F=1-\tfrac{6}{\pi^2}\sum n^{-2}e^{-n^2\pi^2Dt/R^2}$.

\subsection{Status and numerical limits}
\begin{itemize}
\item \textbf{Completeness and truncation.} Orthogonality and the interface and boundary cancellations are derived in the foundation document. Completeness of the eigenfunctions for this singular piecewise problem is not proved there, so the series is verified numerically (Section~\ref{sec:verification}), including the weight sum $\sum_nw_n\to1$. Modes are kept up to $\Lambda_{\max}$ and the series is used only for $t\ge40/\Lambda_{\max}$, where omitted modes are suppressed by $e^{-40}$.
\item \textbf{Conditioning.} When the flux $D(u'-u/r)$ at an interface is much smaller than $D\,u'$ and $D\,u/r$ (strongly contrasting adjacent diffusivities), the propagation loses digits by cancellation. The code reports the smallest value of $|D(u'-u/r)|/(|Du'|+|Du/r|)$ over modes and interfaces, and an accuracy floor of about $2.2\times10^{-16}$ divided by that ratio. Differences between FV and the series below that floor are not resolved.
\end{itemize}

\subsection{Homogeneous Robin sphere (Case B reference)}\label{sec:robin}
For $\phi=\sin(zr/R)/r$ and $\mathrm{Bi}=hR/D$, the condition $-D\phi'(R)=h\phi(R)$ gives
\[
z\cot z=1-\mathrm{Bi}.
\]

\section{Conservative spherical finite volumes}
\subsection{Geometry}
Cell $j$ occupies $r_{j-1/2}<r<r_{j+1/2}$ with interfaces aligned to cell faces.
\begin{equation}
V_j=\frac{4\pi}{3}\left(r_{j+1/2}^3-r_{j-1/2}^3\right),\qquad A_{j\pm1/2}=4\pi r_{j\pm1/2}^2.
\tag{TRISO-FV-101}
\end{equation}
The unknown is the exact cell average $C_j=V_j^{-1}\int c\,4\pi r^2dr$ (TRISO-FV-118) represented at the volume centroid
\begin{equation}
r_j=\frac{3}{4}\,\frac{r_{j+1/2}^4-r_{j-1/2}^4}{r_{j+1/2}^3-r_{j-1/2}^3}.
\tag{TRISO-FV-109A}
\end{equation}

\subsection{Exact cell balance}
Multiply Equation TRISO-FV-100 by $4\pi r^2$ and integrate over the cell. The derivative term integrates exactly, $4\pi\left[r^2D\,\partial_rc\right]_{r_{j-1/2}}^{r_{j+1/2}}$. With $J_r=-D\,\partial_rc$ (TRISO-FV-114) this equals $A_{j-1/2}J_{j-1/2}-A_{j+1/2}J_{j+1/2}$, so
\begin{equation}
V_j\frac{dC_j}{dt}=A_{j-1/2}J_{j-1/2}-A_{j+1/2}J_{j+1/2}+S_jV_j.
\tag{TRISO-FV-123}
\end{equation}
This is exact before any approximation of the face fluxes.

\subsection{Face fluxes}
Inside one material, with $\delta r_{PE}=r_E-r_P$ (TRISO-FV-124 to TRISO-FV-128),
\[
J_e\approx D\,\frac{C_P-C_E}{\delta r_{PE}} .
\]
At a face between materials, let $\delta_P=r_f-r_P$, $\delta_E=r_E-r_f$ and let $C_I$ be the (continuous) interface concentration. Assume $C$ varies linearly over each half cell. The same flux crosses both halves:
\[
J_f=D_P\frac{C_P-C_I}{\delta_P}=D_E\frac{C_I-C_E}{\delta_E}.
\]
Eliminating $C_I$ (TRISO-FV-133 to TRISO-FV-138) gives $C_P-C_E=J_f\left(\delta_P/D_P+\delta_E/D_E\right)$, hence the conductance
\begin{equation}
\boxed{A_fJ_f=G_f\,(C_P-C_E),\qquad G_f=\frac{A_f}{\delta_P/D_P+\delta_E/D_E}}
\tag{TRISO-FV-140}
\end{equation}
For $D_P=D_E$ this is the ordinary two-point form, and for equal half distances it contains the harmonic mean $2D_PD_E/(D_P+D_E)$. The linear half-cell assumption with the face area $A_f$ in both halves is the approximation made by the scheme. The numerical results of Section~\ref{sec:verification} show second-order convergence to the exact solution, including across material interfaces.

\subsection{Semi-discrete equation and conservation}
Substituting Equation TRISO-FV-140 into Equation TRISO-FV-123, with west and east conductances $G_W$ and $G_E$,
\begin{equation}
\boxed{V_j\frac{dC_j}{dt}=G_W\,(C_{j-1}-C_j)+G_E\,(C_{j+1}-C_j)+S_jV_j}
\tag{TRISO-FV-155}
\end{equation}
The flux $G_f(C_E-C_P)$ leaves one cell and enters its neighbour with opposite sign, so all internal fluxes cancel exactly in $\sum_jV_jC_j$.

\subsection{Centre and outer boundary}
The first cell has $r_{1/2}=0$, so its west face area and $G_W$ vanish. No ghost cell is needed.

For the absorbing surface $c(R,t)=0$ and the last centroid $r_N$,
\begin{equation}
\begin{gathered}
G_R=\frac{4\pi R^2D_5}{R-r_N},\qquad \dot M_{\mathrm{rel}}=G_RC_N,\\
V_N\frac{dC_N}{dt}=G_W(C_{N-1}-C_N)-G_RC_N.
\end{gathered}
\tag{TRISO-DEL-030}
\end{equation}

\subsection{Backward Euler}
For time step $\Delta t$,
\[
\begin{aligned}
\frac{V_j}{\Delta t}\left(C_j^{n+1}-C_j^n\right)={}&G_W\left(C_{j-1}^{n+1}-C_j^{n+1}\right)\\
&+G_E\left(C_{j+1}^{n+1}-C_j^{n+1}\right)+S_j^{n+1}V_j,
\end{aligned}
\]
\begin{equation}
\boxed{
\begin{aligned}
&-G_WC_{j-1}^{n+1}+\left(\frac{V_j}{\Delta t}+G_W+G_E\right)C_j^{n+1}-G_EC_{j+1}^{n+1}\\
&\qquad=\frac{V_j}{\Delta t}C_j^n+S_j^{n+1}V_j
\end{aligned}}
\tag{TRISO-DEL-040}
\end{equation}
The matrix is tridiagonal with positive diagonal and non-positive off-diagonal entries and is strictly diagonally dominant, hence an M-matrix. For non-negative old data and source the new concentrations are non-negative.

\textbf{Subtraction-free elimination.} Let $v_j=V_j/\Delta t$ and $g_j^W=G_W$ of cell $j$ (so $g^W_{j+1}=g^E_j$). Define $e_1=v_1$ and $e_j=v_j+g^W_je_{j-1}/(e_{j-1}+g^W_j)$, pivots $d_j=e_j+g^E_j$, forward sweep $y_j=(\mathrm{rhs}_j+g^W_jy_{j-1})/d_j$, and back substitution $C_j=y_j+(g^E_j/d_j)\,C_{j+1}$ with $C_N=y_N$. Every operation adds, multiplies or divides non-negative numbers. The textbook pivot $b_j-a_jc_{j-1}'$ subtracts nearly equal numbers when the Fourier number $D\Delta t/\Delta r^2$ is large and degrades the inventory balance to about $10^{-9}$. The form above keeps it below $10^{-12}$ (observed $\le3\times10^{-13}$).

For verification only, Forward Euler has the coefficient-positivity bound
\begin{equation}
\boxed{\Delta t\le\min_j\frac{V_j}{G_W+G_E}}
\tag{TRISO-FV-300}
\end{equation}
Backward Euler is the production scheme and has no such restriction.

\subsection{Inventory, release, balance}
\[
\begin{gathered}
M^n=\sum_jV_jC_j^n,\qquad M_{\mathrm{rel}}^{n+1}=M_{\mathrm{rel}}^n+\Delta t\,G_RC_N^{n+1},\\
F^n=\frac{M_{\mathrm{rel}}^n}{M^0},\qquad \dot F^n=\frac{G_RC_N^n}{M^0}.
\end{gathered}
\]
Summing Equation TRISO-DEL-040 over $j$ cancels all internal conductances and leaves
\[
M^{n+1}+M_{\mathrm{rel}}^{n+1}=M^0
\]
up to linear-solver and rounding error. The release uses the same boundary flux as the balance, so the residual $M^n+M_{\mathrm{rel}}^n-M^0$ is an independent diagnostic.

\section{Limitations}
\begin{itemize}
\item Synthetic inputs are not experimental or physical validation.
\item Temperature and fluence dependence of $D$, source schedules and a production Robin release mode are not implemented.
\item The exact series is trustworthy only above its reported accuracy floor. The floor grows roughly in proportion to the diffusivity contrast between adjacent layers. In the stress tests it stayed below $10^{-7}$ up to a contrast of $10^{5}$ and reached $10^{-6}$ at $10^{6}$.
\item Completeness of the eigenfunction set for the singular piecewise problem is checked numerically, not proved.
\item WOS and Process-C results are separate verification tracks and are not evidence of FV or WOS equivalence.
\end{itemize}
